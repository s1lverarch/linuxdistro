      // pacman
      
      Add the SilverNebula repository
      
      
      Import and verify the key first, then add the repository with signatures required.
      
    
      
      
      
        STEP 1
        
      Import and verify the key
      
        
      Download the public key and check its fingerprint.
      
        
      terminal 
      
          
      curl -fLO https://s1lverarch.github.io/linuxdistro/keys/silvernebula.gpg
gpg --show-keys --with-fingerprint silvernebula.gpg
sudo pacman-key --add silvernebula.gpg
sudo pacman-key --lsign-key 8EA07B2205012BA364DBB14BCC892E251DCD1C44
      
        
      8EA0 7B22 0501 2BA3 64DB B14B CC89 2E25 1DCD 1C44 · If the fingerprint differs, stop.
      
      
      
      
      
        STEP 2
        
      Add the repository, signed
      
        
      Append this block to /etc/pacman.conf.
      
        
      pacman.conf 
      
          
      [silvernebula]
SigLevel = Required DatabaseOptional
Server   = https://raw.githubusercontent.com/s1lverarch/silvernebula/main/$arch
      
      
      
      
      
        STEP 3
        
      Sync and install the keyring
      
        
      The keyring package keeps the keys updated.
      
        
      terminal 
      
          
      sudo pacman -Syu
sudo pacman -S silvernebula-keyring
